# params.txt — 逐项对照参考仓库 dev/gfold/params.txt
#
# 【原则】结构与参数名尽量与参考仓库一致，便于对照。
#   凡是参考仓库有的，本文件沿用同名同义；只改【载具相关数值】。
#   参考仓库原文见 dev/gfold/params.txt（本文件在每行末尾标注对应关系）。

# ---------- 环境参数 ----------
g0 = 9.807          # 参考同值

# 【本项目差异】参考仓库 start_altitude = 1100（它的用例从 1.1 km 开始规划）。
#   本项目交班高度门 gfold_h = 5000 m（kOS 段1 交给 kRPC 的高度）。
start_altitude = 5000

# ---------- 求解条件 ----------
# 参考 tf = 20。本项目必须更大：实飞交班点横速 361 m/s，
#   tf=20 时推力锥咬死导致 P3 infeasible（实测扫描，见 gfold_p3p4.solve_p3p4 文档）。
#   tf 只是 P3 的【上限】，真正落地时刻由轨迹读出，给大是安全的。
tf = 40
straight_fac = 5    # 参考同值
direct = True       # 本项目用 cvxpy 直接求解（未编译 cvxpy_codegen 扩展）

# ---------- 目标参数 ----------
target_lat = -0.0972060951675948    # 【本项目】B1040-9 回收场（与 kOS tgt_lat 一致）
target_lon = -74.5576822740041      # 【本项目】同上
target_height = 36.8                # 【本项目】着陆腿触点高度（kOS target_alt 同值）

# ---------- 限制参数 ----------
max_tilt = 25       # 参考同值。推力轴最大倾角，用于 conic_clamp
throttle_limit = [0.1, 0.8]         # 参考同值。规划时节流限制
throttle_limit_ctrl = [0.05, 1.0]   # 参考同值。实际执行时节流限制
G_max = 10          # 参考同值
V_max = 150         # 参考同值。注意：这是【求解器内部】速度上限。
                    # 本项目交班速度可达 478 m/s > 150！
                    # 参考仓库用在 1.1 km 起始、速度已降下来的场景。
                    # 本项目见下方 v_max_solver 说明。
y_gs = 30           # 参考同值。滑翔角锥半角

# 【本项目补充】求解器速度上限必须 >= 入场速度，否则 node0 自相矛盾
#   （要求同时等于初值又 <= V_max）。参考仓库的 V_max=150 在本项目
#   会把可行域错误地卡死，故单独给出。
v_max_solver = 1200

# 推力指向锥：参考仓库用固定 p_cs = max_tilt*0.85。
#   本项目入场横速大，全程固定会 infeasible（坑①），改为按时序收紧。
pcs_start_deg = 85  # 起始宽（允许大倾角制动）
pcs_end_deg = 30    # 末端窄（保证触地姿态）

# 求解器节点数（参考 GFOLD_params.py: N3=160, N4=80）
N3 = 160
N4 = 80

# ---------- 姿态控制 PID（参考仓库同值）----------
ctrl_x_rot.kp = 5
ctrl_x_rot.kd = 2.5
ctrl_y_avel_kp = 2      # y 为滚转
ctrl_z_rot.kp = 5
ctrl_z_rot.kd = 2.5

# ---------- 滑行中调整控制（参考仓库同值）----------
k_x = 0.5
k_v = 0.8

# ---------- 最终降落段（参考仓库同值）----------
final_height = 200      # 触发区域高度
final_radius = 20       # 触发区域半径
final_throttle = 0.8
final_kp = 1

# ---------- 载具参数（本项目实测）----------
isp = 315               # 比冲 [s]
t_max = 8990000         # 最大推力 [N]（实飞日志 8.99 MN）

# ---------- debug ----------
debug_lines = False     # 是否画参考轨迹线（参考仓库默认 True）
